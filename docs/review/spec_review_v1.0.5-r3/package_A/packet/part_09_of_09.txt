def frame_files(repo_root: Path) -> list[str]:
    out = []
    for p in python_files(Path(repo_root)):
        t = p.read_text(encoding="utf-8", errors="replace")
        if PYCA.search(t) or PYCRYPTODOME.search(t):
            out.append(p.relative_to(repo_root).as_posix())

def frame_hash(frames: dict[str, list[str]]) -> str:
    return hashlib.sha256(json.dumps({r: sorted(v) for r, v in frames.items()}, sort_keys=True).encode()).hexdigest()


def worklist(frames: dict[str, list[str]]) -> list[tuple[str, str]]:
    """Round-robin over repositories (ascending name), each repository's files in a seeded order."""
    orders = {r: sorted(v, key=lambda f: hashlib.sha256(f"{SEED}|{r}|{f}".encode()).hexdigest()) for r, v in frames.items()}
    out, i = [], 0
    while any(i < len(v) for v in orders.values()):
        for r in sorted(orders):
            if i < len(orders[r]):
                out.append((r, orders[r][i]))
        i += 1
    return out


def stop_reached(audited: Sequence[dict], repos: Sequence[str]) -> bool:
    """audited: [{"repo", "file", "eligible_lines": [...]}] in worklist order; repos: EVERY repository of the audit
    frame. Stop only when >= MIN_ELIGIBLE eligible sites have been audited AND every repository of the frame (not only
    those already visited) has contributed >= MIN_FILES_PER_REPO files (protocol v1.0.5 Sec. 2; fixed 2026-11-18:
    the earlier version checked only the repositories already audited)."""
    if not repos:
        raise ValueError("the audit frame's repositories must be given")
    return sum(len(a["eligible_lines"]) for a in audited) >= MIN_ELIGIBLE and all(
        sum(a["repo"] == r for a in audited) >= MIN_FILES_PER_REPO for r in repos)


def membership(audited: Sequence[dict], pools: dict[str, set[tuple[str, int]]], sources: dict[str, str]) -> list[dict]:
    """Per audited file: eligible sites found by the owner and whether each is in the candidate pool (statement level).
    `sources` maps (repo) -> repository root, used to map lines to statement starts."""
    out = []
    for a in audited:
        text = (Path(sources[a["repo"]]) / a["file"]).read_text(encoding="utf-8", errors="replace")
        starts = statement_starts(text)
        found = []
        for ln in a["eligible_lines"]:
            st = starts.get(int(ln), int(ln))
            found.append({"line": int(ln), "statement": st, "in_pool": (a["file"], st) in pools[a["repo"]]})
        out.append({"repo": a["repo"], "file": a["file"], "sites": found})

def missed_share(per_file: Sequence[dict], *, B: int = 2_000, seed: int = 20261113) -> dict:
    """Share of audited eligible sites missing from the pool, with a file-level cluster bootstrap 95% interval."""
    files = [(f["repo"], len(f["sites"]), sum(not s["in_pool"] for s in f["sites"])) for f in per_file]
    tot = sum(n for _, n, _ in files)
    miss = sum(m for _, _, m in files)
    if tot == 0:
        return {"eligible": 0, "missed": 0, "share": None, "ci95": (None, None)}
    by_repo = defaultdict(list)
    for r, n, m in files:
        by_repo[r].append((n, m))
    rng = np.random.default_rng(seed)
    shares = []
    for _ in range(B):
        nn = mm = 0
        for v in by_repo.values():
            for k in rng.integers(0, len(v), len(v)):
                nn += v[k][0]
                mm += v[k][1]
        shares.append(mm / nn if nn else 0.0)
    return {"eligible": tot, "missed": miss, "share": miss / tot, "files": len(files),
            "ci95": (float(np.quantile(shares, 0.025)), float(np.quantile(shares, 0.975))),
            "can_estimate": "share of eligible sites in crypto-importing files of the 6 development/validation repositories missed by the pool",
            "cannot_estimate": "recall outside such files, in the test split, or in Python projects generally; errors shared by the owner and the pool"}

"""Multi-call candidate statements (owner decision F03; annotation guide v1.2 Step A "Multi-call candidate statement").

The candidate pool is deduplicated by statement (protocol V6, unchanged), while a site is one call expression (role
taxonomy rule 7). For the few candidate statements that contain more than one call to a supported cryptographic API,
the guide fixes which call is the candidate's site. This module only LISTS the calls of a statement, by API name and in
source order, so that the issued to-do entry can be marked `multi_call` and the owner can apply the rule. It never
decides a role, eligibility or the site: those stay with the owner. It reads source text only, never tool output.

Names (receiver types are not resolved, so a builder method called `public_key` is listed too):
  operation:    sign, verify, exchange, encrypt, decrypt

                from_encoded_point, private_numbers, public_numbers, load_pem_* / load_der_* / load_ssh_*,

from __future__ import annotations

import ast

OPERATIONS = frozenset({"sign", "verify", "exchange", "encrypt", "decrypt"})
KEY_MATERIAL = frozenset({"generate_private_key", "import_key", "construct", "public_key", "from_private_bytes",
                          "from_public_bytes", "from_encoded_point", "private_numbers", "public_numbers"})
KEY_PREFIXES = ("load_pem_", "load_der_", "load_ssh_")
GENERATE_RECEIVERS = frozenset({"RSA", "ECC", "DSA"})
CONSTRUCTOR_RECEIVERS = frozenset({"PKCS1_OAEP", "PKCS1_v1_5", "pkcs1_15", "pss", "DSS"})


def call_kind(call: ast.Call) -> str | None:
    """'operation' / 'key_material' / 'constructor' for a supported API name, else None."""
    f = call.func
    if isinstance(f, ast.Attribute):
        name, recv = f.attr, (f.value.id if isinstance(f.value, ast.Name) else None)
    elif isinstance(f, ast.Name):
        name, recv = f.id, None
    else:
        return None
    if name in OPERATIONS:
        return "operation"
    if name in KEY_MATERIAL or name.startswith(KEY_PREFIXES) or (name == "generate" and recv in GENERATE_RECEIVERS):
        return "key_material"
    if name == "new" and recv in CONSTRUCTOR_RECEIVERS:
        return "constructor"
    return None


def _statement_at(tree: ast.AST, line: int) -> ast.stmt | None:
    best = None
    for n in ast.walk(tree):
        if isinstance(n, ast.stmt) and n.lineno == line:
            if best is None or (n.end_lineno - n.lineno) < (best.end_lineno - best.lineno):
                best = n
    return best


def statement_calls(text: str, line: int) -> list[dict] | None:
    """Supported-API calls of the statement that starts at `line` (header only for compound statements), in source
    order (line, column of the called name). None if the file does not parse or no statement starts there."""
    try:
        tree = ast.parse(text)
    except (SyntaxError, ValueError):
        return None
    node = _statement_at(tree, line)
    if node is None:
        return None
    body = {id(x) for fld in ("body", "orelse", "finalbody", "handlers")
            for c in (getattr(node, fld, None) or []) for x in ast.walk(c)}
    out = []
    for c in ast.walk(node):
        if isinstance(c, ast.Call) and id(c) not in body:
            k = call_kind(c)
            if k:
                f = c.func
                name = f.attr if isinstance(f, ast.Attribute) else f.id
                pos = (f.end_lineno, f.end_col_offset - len(name)) if isinstance(f, ast.Attribute) else (f.lineno, f.col_offset)
                out.append({"line": pos[0], "col": pos[1], "name": name, "kind": k})
    return sorted(out, key=lambda d: (d["line"], d["col"], d["name"]))


def mark(text: str, line: int) -> dict:
    """Fields added to an issued to-do entry: {'multi_call': bool, 'calls': [...]} (calls only when multi_call)."""
    calls = statement_calls(text, line) or []
    return {"multi_call": True, "calls": calls} if len(calls) > 1 else {"multi_call": False}

"""H2 inference (phase3_analysis_plan.md Sec. 2, 5, 7, 8; decisions V10a, V10b, V13, V20, V21; R4-19 items 10-12).

Input: per comparison, the per-repository results of `pqscan.bench.matching.repository_result` for the seven test
repositories. Output: the estimates, bootstrap p-values, guard, gate, K/K_e assessability, Holm (m = 3) and the
outcome statements. Nothing here reads labels directly: correctness enters only through the matched arrays.

Bootstrap (Sec. 7.1): for b = 1..B, within each estimable repository (ascending name), draw n_r indices with
replacement and n_r uniform tie keys per arm; recount, re-match lowest tier first (ties by the keys), re-score and
pool with the point estimate's weighting. p(d0) = (1 + #{delta* <= d0}) / (B + 1). The repository-by-stratum
variant (Sec. 2 fallback) resamples within (repository x stratum) cells instead; it is used only if the pre-F1
two-stratum calibration requires it (BOOTSTRAP_VARIANT below records which variant is in force).
"""
from __future__ import annotations

import math
from fractions import Fraction
from typing import Sequence

import numpy as np
from scipy import stats as sps

from .stats import holm

ALPHA = 0.025
MARGIN = 0.10
GUARD = Fraction(1, 10)
K_MIN, C_MIN, FEASIBLE_MIN, K_REPOS = 5, Fraction(1, 5), 5, 7
B_PRIMARY, B_RERUN, SEED = 10_000, 100_000, 20261113
CONFIRMATORY = ("H2-1", "H2-3", "H2-4")
BOOTSTRAP_VARIANT = "unstratified"          # "stratified" only if the pre-F1 two-stratum calibration fails (Sec. 11)


# --------------------------------------------------------------------------------------------- bootstrap
def _withdraw_mask(rec: np.ndarray, tier: np.ndarray, keys: np.ndarray, excess: np.ndarray) -> np.ndarray:
    """rec, tier, keys: (b, n); excess: (b,). True for the `excess` recommendations with the smallest tier + key."""
    score = np.where(rec, tier.astype(float) + keys, np.inf)
    ranks = np.argsort(np.argsort(score, axis=1, kind="stable"), axis=1, kind="stable")
    return rec & (ranks < excess[:, None])


def resampled_delta(F: tuple, M: tuple, idx: np.ndarray, keys_f: np.ndarray, keys_m: np.ndarray) -> np.ndarray:
    """Matched difference per resample of one repository (random tie keys inside the resample)."""
    rf, tf, kf, wf = (a[idx] for a in F)
    rm, tm, km, wm = (a[idx] for a in M)
    nf, nm = rf.sum(axis=1), rm.sum(axis=1)
    k = np.minimum(nf, nm)
    xf = np.where(_withdraw_mask(rf, tf, keys_f, nf - k), wf, kf).sum(axis=1)
    ym = np.where(_withdraw_mask(rm, tm, keys_m, nm - k), wm, km).sum(axis=1)
    return (xf - ym) / idx.shape[1]


def _indices(rng, n: int, b: int, strata: Sequence | None):
    if strata is None:
        return rng.integers(0, n, size=(b, n))
    strata = np.asarray(strata, dtype=object)
    idx = np.empty((b, n), dtype=np.int64)
    for s in sorted(set(strata.tolist()), key=str):
        pos = np.flatnonzero(strata == s)
        idx[:, pos] = pos[rng.integers(0, len(pos), size=(b, len(pos)))]
    return idx


def bootstrap(repos: Sequence[tuple[tuple, tuple]], *, B: int = B_PRIMARY, seed: int = SEED, chunk: int = 2_000,
              weighting: str = "repository", strata: Sequence[Sequence] | None = None) -> np.ndarray:
    """Distribution of delta* (length B). `repos` are (F arrays, M arrays) of the estimable repositories in ascending
    repository-name order. `strata` (per repository, per site) selects the repository-by-stratum variant."""
    rng = np.random.default_rng(seed)
    ns = np.array([F[0].size for F, _ in repos], dtype=float)
    out = np.empty(B)
    done = 0
    while done < B:
        b = min(chunk, B - done)
        vals = np.empty((b, len(repos)))
        for j, (F, M) in enumerate(repos):
            n = F[0].size
            idx = _indices(rng, n, b, None if strata is None else strata[j])
            vals[:, j] = resampled_delta(F, M, idx, rng.random((b, n)), rng.random((b, n)))
        out[done:done + b] = vals.mean(axis=1) if weighting == "repository" else (vals * ns).sum(axis=1) / ns.sum()
        done += b
    return out


def p_value(dist: np.ndarray, d0: float) -> float:
    return (1 + int(np.count_nonzero(dist <= d0 + 1e-12))) / (dist.size + 1)


def mc_rerun_needed(p: float, threshold: float, B: int) -> bool:
    return abs(p - threshold) < 3 * math.sqrt(max(p * (1 - p), 1e-300) / B)

def coverage_gap(results: Sequence[dict]) -> Fraction | None:
    """g = (1/K_e) sum over estimable repositories of (R_F - R_M)/n_r, exact. None when K_e = 0 (undefined)."""
    est = [r for r in results if r.get("n_common", 0) >= 1]
    if not est:
        return None
    return sum((Fraction(r["R_f"] - r["R_m"], r["n_common"]) for r in est), Fraction(0)) / len(est)


def guard_active(g: Fraction | None) -> bool:
    if g is None:
        raise ValueError("the coverage gap is undefined (K_e = 0)")
    return g < GUARD


def guarded(p: float, g: Fraction) -> float:
    return min(1.0, 2.0 * p) if guard_active(g) else p


def gate(results: Sequence[dict]) -> dict:
    est = [r for r in results if r.get("n_common", 0) >= 1]
    k = [r["k"] for r in est]
    n = [r["n_common"] for r in est]
    feasible = [r["repo"] for r in results if r.get("n_common", 0) >= 1 and r["k"] >= K_MIN]
    c_pool = Fraction(sum(k), sum(n)) if sum(n) else None
    c_bar = (sum(Fraction(a, b) for a, b in zip(k, n)) / len(n)) if n else None
    return {"c_pool": c_pool, "c_bar": c_bar, "feasible": feasible, "n_feasible": len(feasible),
            "passes": bool(c_pool is not None and c_pool >= C_MIN and len(feasible) >= FEASIBLE_MIN)}

    if K_e == 0:

def analyse_comparison(name: str, results: Sequence[dict], *, B: int = B_PRIMARY, seed: int = SEED,

    K = len(results)
    est = [r for r in results if r.get("n_common", 0) >= 1]
    K_e = len(est)
    out = {"comparison": name, "K": K, "K_e": K_e, "assessability": assessability(K, K_e),
           "non_estimable": [{"repo": r["repo"], "n_eligible": r["n_eligible"], "partition": r["partition"]}
                             for r in results if r.get("n_common", 0) < 1],
           "gate": gate(results), "B": B, "seed": seed, "variant": variant}
    if K_e == 0:
        out.update(g=None, guard=None, delta=None, delta_site=None, p=None, p_guarded=None, lower_975=None)

    g = coverage_gap(results)
    deltas = [r["delta_ta"] for r in est]

               lower_975=float(np.quantile(dist, 0.025)), lower_9875=float(np.quantile(dist, 0.0125)),
               dist_mean=float(dist.mean()), dist_sd=float(dist.std(ddof=1)),

def holm_thresholds(pvals: dict[str, float]) -> dict[str, float]:
    order = sorted(pvals, key=lambda c: pvals[c])
    return {c: ALPHA / (len(order) - i) for i, c in enumerate(order)}


def run_family(results_by_comparison: dict[str, Sequence[dict]], *, B: int = B_PRIMARY, seed: int = SEED,
               variant: str = BOOTSTRAP_VARIANT, mc_rule: bool = True) -> dict:
    """Steps 3-10 of plan Sec. 7.5 for H2-1, H2-3, H2-4 (H2-S analysed alongside, outside the family).
    `mc_rule=False` is used only by the pre-F1 calibration, which measures the procedure at a fixed B (the rule only
    reduces the Monte Carlo error of p); the real analysis always applies it."""
    res = {c: analyse_comparison(c, results_by_comparison[c], B=B, seed=seed, variant=variant)
           for c in results_by_comparison}
    # Monte Carlo rule: re-run a comparison with B = 100,000 if its guarded p is within 3 MC-SE of its threshold
    def guarded_or_one(c):
        r = res[c]
        return r["p_guarded"] if r["assessability"] == "assessable" else 1.0
    fam = [c for c in CONFIRMATORY if c in res]
    for _ in range(2 if mc_rule else 0):
        thr = holm_thresholds({c: guarded_or_one(c) for c in fam})
        rerun = [c for c in fam if res[c]["assessability"] == "assessable" and res[c]["B"] < B_RERUN
                 and mc_rerun_needed(res[c]["p"], thr[c] / (2 if res[c]["guard"] else 1), res[c]["B"])]
        if not rerun:
            break
        for c in rerun:
            res[c] = analyse_comparison(c, results_by_comparison[c], B=B_RERUN, seed=seed, variant=variant)
            res[c]["mc_rerun"] = True
    pstar = {c: guarded_or_one(c) for c in fam}
    adj = dict(zip(fam, holm([pstar[c] for c in fam]))) if fam else {}
    for c in fam:
        r = res[c]
        r["p_holm"] = adj[c]
        r["holm_rejected"] = adj[c] <= ALPHA
        r["claimable"] = r["assessability"] == "assessable" and r["holm_rejected"] and r["gate"]["passes"]
        r["statement"] = statement(c, r)
    if "H2-1" in res:
        r = res["H2-1"]
        r["h2b"] = h2b_statement(r)
    if "H2-S" in res:
        res["H2-S"]["statement"] = "secondary (outside the family)"
    return res


def statement(c: str, r: dict) -> str:
    """Plan Sec. 8. Never 'H2 supported'."""
    label = {"H2-1": "H2a (confirmatory)", "H2-3": "H2-3 (confirmatory)", "H2-4": "H2-4 (confirmatory)"}[c]
    if r["assessability"] != "assessable":
        return f"{label}: {r['assessability']}"
    if not r["gate"]["passes"]:
        return f"{label}: not assessable at equal coverage (matched coverage too low)"
    return f"{label}: {'supported' if r['claimable'] else 'not supported'}"


def h2b_statement(r: dict) -> str:
    if r["assessability"] != "assessable":
        return "H2b (exploratory, pre-specified; not under family-wise error control): not assessable"
    met = r["p_margin_guarded"] <= ALPHA
    return ("H2b (exploratory, pre-specified; not under family-wise error control): 10-point criterion "
            + ("met" if met else "not met"))

def repository_level(deltas: Sequence[float]) -> dict:
    """Exploratory generalisation analyses on the per-repository estimates (labelled exploratory; R4-04)."""
    d = np.asarray(deltas, dtype=float)
    K = d.size
    out = {"K_e": K, "deltas": d.tolist(), "leave_one_out": [float(np.delete(d, i).mean()) for i in range(K)] if K > 1 else []}
    if K >= 2 and d.std(ddof=1) > 0:
        out["t_one_sided_p"] = float(sps.ttest_1samp(d, 0.0, alternative="greater").pvalue)
    if K >= 1:
        signs = np.array(np.meshgrid(*[[-1, 1]] * K)).reshape(K, -1).T
        stat = (signs * np.abs(d)).sum(axis=1)
        out["sign_flip_p"] = float(np.mean(stat >= d.sum() - 1e-12))
        if np.any(d != 0):
            out["wilcoxon_one_sided_p"] = float(sps.wilcoxon(d, alternative="greater").pvalue)
    return out

"""Descriptive repository-gap diagnostics for the coverage-gap guard (analysis plan v1.0.1-r2 Sec. 7.2; second review G10).

The pre-specified guard (pqscan.bench.h2.coverage_gap / guard_active) uses the SIGNED, repository-balanced gap
g = (1/K_e) sum_r (R_F,r - R_M,r) / n_r. These diagnostics are reported next to it and NEVER change the guard, p or
any decision. They show what the signed average can hide: opposite-direction repositories (mean absolute gap) and the
repository where F's coverage advantage is smallest (minimum gap). This module is outside the E30 calibration closure.
"""
from __future__ import annotations

from fractions import Fraction
from typing import Mapping, Sequence


def repository_gaps(results: Sequence[Mapping]) -> list[Fraction]:
    """(R_F,r - R_M,r) / n_r for every estimable repository (n_common >= 1), as exact fractions."""
    return [Fraction(int(r["R_f"]) - int(r["R_m"]), int(r["n_common"])) for r in results if int(r["n_common"]) >= 1]


def diagnostics(results: Sequence[Mapping]) -> dict:
    gaps = repository_gaps(results)
    if not gaps:
        return {"K_e": 0, "signed_mean": None, "mean_absolute": None, "minimum": None, "maximum": None,
                "opposite_direction_repositories": 0, "descriptive_only": True}
    signed = sum(gaps, Fraction(0)) / len(gaps)
    return {"K_e": len(gaps), "signed_mean": signed, "mean_absolute": sum(map(abs, gaps), Fraction(0)) / len(gaps),
            "minimum": min(gaps), "maximum": max(gaps),
            "opposite_direction_repositories": sum(1 for x in gaps if (x < 0) != (signed < 0) and x != 0),
            "descriptive_only": True}
